cmake_minimum_required(VERSION 3.10)
project(px4_v1_12_3_reference LANGUAGES CXX)
set(upstream "${CMAKE_CURRENT_SOURCE_DIR}/upstream")
add_library(px4_v1_12_3 STATIC
  ${upstream}/src/modules/mc_att_control/AttitudeControl/AttitudeControl.cpp
  ${upstream}/src/modules/mc_rate_control/RateControl/RateControl.cpp
  ${upstream}/src/modules/mc_pos_control/PositionControl/PositionControl.cpp
  ${upstream}/src/modules/mc_pos_control/PositionControl/ControlMath.cpp
  ${upstream}/src/lib/mixer/MixerBase/Mixer.cpp
  ${upstream}/src/lib/mixer/MultirotorMixer/MultirotorMixer.cpp)
target_compile_features(px4_v1_12_3 PUBLIC cxx_std_17)
set_target_properties(px4_v1_12_3 PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(px4_v1_12_3 SYSTEM PUBLIC
  ${CMAKE_CURRENT_SOURCE_DIR}/include
  ${upstream}/src
  ${upstream}/src/lib
  ${upstream}/src/include
  ${upstream}/platforms/common/include
  ${upstream}/src/modules/mc_att_control/AttitudeControl
  ${upstream}/src/modules/mc_rate_control/RateControl
  ${upstream}/src/modules/mc_pos_control/PositionControl)
add_library(px4_v1_12_3::core ALIAS px4_v1_12_3)
# Generated by the original px_generate_mixers.py from pinned quad_x.toml.
# The physical plant may supply its own immutable Rotor array to the original
# array constructor; no geometry or controller parameters are invented here.
option(PX4_REFERENCE_BUILD_TESTS "Build the offline oracle and original upstream tests" OFF)
if(PX4_REFERENCE_BUILD_TESTS)
  add_executable(px4_reference_test ../px4_reference_test.cpp)
  target_link_libraries(px4_reference_test PRIVATE px4_v1_12_3)
  enable_testing()
  add_test(NAME px4_reference_history COMMAND px4_reference_test)
  find_package(GTest REQUIRED)
  find_package(Threads REQUIRED)
  add_executable(px4_upstream_tests
    ${upstream}/src/modules/mc_rate_control/RateControl/RateControlTest.cpp
    ${upstream}/src/modules/mc_pos_control/PositionControl/PositionControlTest.cpp
    ${upstream}/src/modules/mc_pos_control/PositionControl/ControlMathTest.cpp)
  target_include_directories(px4_upstream_tests SYSTEM PRIVATE ${GTEST_INCLUDE_DIRS})
  target_link_libraries(px4_upstream_tests PRIVATE px4_v1_12_3 ${GTEST_BOTH_LIBRARIES} Threads::Threads)
  enable_testing()
  add_test(NAME px4_upstream_original_tests COMMAND px4_upstream_tests)
endif()
