cmake_minimum_required(VERSION 3.16)
project(xgc_lightweight_vehicle LANGUAGES CXX)

include(GNUInstallDirs)
set(XGC2_MATH_INCLUDE "/usr/include" CACHE PATH "Installed xgc2-math headers or explicit source include")
set(EIGEN_INCLUDE "/usr/include/eigen3" CACHE PATH "Eigen headers")
# Installed owner exports only: SDK3 carries runtime primitives, while
# generic robotics and simulation DTOs are consumed from their public owners.
find_package(XgcRuntimeSDK CONFIG REQUIRED)
find_package(XgcRoboticsInterfaces CONFIG REQUIRED)
find_package(XgcLightweightSimInterfaces CONFIG REQUIRED)
set(FS150_ASSET_SOURCE_ROOT "" CACHE PATH "Owning FS150 SITL asset source")
if(NOT EXISTS "${FS150_ASSET_SOURCE_ROOT}/scripts/generate_native_flight_model.py")
  message(FATAL_ERROR "Set FS150_ASSET_SOURCE_ROOT; no copied physical defaults")
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/generated")
set(asset_header "${CMAKE_CURRENT_BINARY_DIR}/generated/fs150_native_flight_model.hpp")
set(asset_json "${CMAKE_CURRENT_BINARY_DIR}/generated/fs150_native_flight_model.json")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${FS150_ASSET_SOURCE_ROOT}/scripts/generate_native_flight_model.py"
  --sdf "${FS150_ASSET_SOURCE_ROOT}/models/fs150/iris.sdf"
  --params "${FS150_ASSET_SOURCE_ROOT}/config/generated/fs150-sitl.params"
  --fcu-name-catalog "${CMAKE_CURRENT_SOURCE_DIR}/fcu_parameter_names.json"
  --output "${asset_json}" --cpp-output "${asset_header}" RESULT_VARIABLE asset_status)
if(NOT asset_status EQUAL 0)
  message(FATAL_ERROR "FS150 asset extraction failed")
endif()
include_directories("${CMAKE_CURRENT_BINARY_DIR}/generated")
option(LIGHTWEIGHT_TESTS "Build the model and native boundary checks" OFF)

add_subdirectory(px4_v1_12_3)
add_library(lightweight_vehicle SHARED lightweight_vehicle.cpp)
target_compile_features(lightweight_vehicle PRIVATE cxx_std_17)
target_compile_options(lightweight_vehicle PRIVATE -O2 -Wall -Wextra -Werror)
set_target_properties(lightweight_vehicle PROPERTIES CXX_VISIBILITY_PRESET hidden)
target_include_directories(lightweight_vehicle PRIVATE
  "${XGC2_MATH_INCLUDE}")
target_include_directories(lightweight_vehicle SYSTEM PRIVATE "${EIGEN_INCLUDE}")
target_link_libraries(lightweight_vehicle PRIVATE px4_v1_12_3::core XgcRuntime::SDK XgcRoboticsInterfaces::Interfaces XgcLightweightSim::Interfaces)

if(LIGHTWEIGHT_TESTS)
  enable_testing()
  foreach(test_name rigid_body flight_controller model px4_setpoint ground_contact)
    add_executable(${test_name}_test ${test_name}_test.cpp)
    target_compile_features(${test_name}_test PRIVATE cxx_std_17)
    target_compile_options(${test_name}_test PRIVATE -O2 -Wall -Wextra -Werror -UNDEBUG)
    target_include_directories(${test_name}_test PRIVATE "${XGC2_MATH_INCLUDE}")
    target_include_directories(${test_name}_test SYSTEM PRIVATE "${EIGEN_INCLUDE}")
    target_link_libraries(${test_name}_test PRIVATE px4_v1_12_3::core XgcRuntime::SDK XgcRoboticsInterfaces::Interfaces XgcLightweightSim::Interfaces)
    add_test(NAME lightweight_${test_name} COMMAND ${test_name}_test)
  endforeach()
  add_executable(abi_test abi_test.cpp lightweight_vehicle.cpp)
  target_compile_features(abi_test PRIVATE cxx_std_17)
  target_compile_options(abi_test PRIVATE -O2 -Wall -Wextra -Werror -UNDEBUG)
  target_include_directories(abi_test PRIVATE
    "${XGC2_MATH_INCLUDE}")
  target_include_directories(abi_test SYSTEM PRIVATE "${EIGEN_INCLUDE}")
  target_link_libraries(abi_test PRIVATE px4_v1_12_3::core XgcRuntime::SDK XgcRoboticsInterfaces::Interfaces XgcLightweightSim::Interfaces)
  add_test(NAME lightweight_abi COMMAND abi_test)
endif()

option(LIGHTWEIGHT_CONTROLLER_TEST "Build the installed native controller fixture" OFF)
if(LIGHTWEIGHT_CONTROLLER_TEST)
  find_package(HoverThrustNativeWire 1.0 CONFIG REQUIRED)
  find_package(ReferenceTrajectoryNative CONFIG REQUIRED)
  find_package(Threads REQUIRED)
  add_executable(controller_test controller_test.cpp)
  target_compile_features(controller_test PRIVATE cxx_std_17)
  target_compile_options(controller_test PRIVATE -O2 -Wall -Wextra -Werror -UNDEBUG)
  target_link_libraries(controller_test PRIVATE XgcRuntime::SDK XgcRoboticsInterfaces::Interfaces XgcLightweightSim::Interfaces HoverThrustNative::Wire ReferenceTrajectoryNative::Wire Threads::Threads ${CMAKE_DL_LIBS})
endif()
if(LIGHTWEIGHT_TESTS)
  add_executable(provider_test provider_test.cpp)
  target_compile_features(provider_test PRIVATE cxx_std_17)
  target_compile_options(provider_test PRIVATE -O2 -Wall -Wextra -Werror -UNDEBUG)
  target_link_libraries(provider_test PRIVATE XgcRuntime::SDK XgcRoboticsInterfaces::Interfaces XgcLightweightSim::Interfaces ${CMAKE_DL_LIBS})
  add_test(NAME lightweight_provider COMMAND provider_test $<TARGET_FILE:lightweight_vehicle>)
endif()

install(TARGETS lightweight_vehicle LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}/xgc2-lightweight-sim)
install(FILES "${asset_json}" DESTINATION ${CMAKE_INSTALL_DATADIR}/xgc2-lightweight-sim)
