cmake_minimum_required(VERSION 3.16)
project(xsim LANGUAGES C CXX)

include(GNUInstallDirs)
set(XGC2_MATH_INCLUDE "/usr/include" CACHE PATH "Installed xgc2-math headers or explicit source include")
set(EIGEN_INCLUDE "/usr/include/eigen3" CACHE PATH "Eigen headers")
# The server consumes public robotics records; no runtime SDK or Host.
find_package(XgcRoboticsInterfaces CONFIG REQUIRED)
set(FS150_ASSET_SOURCE_ROOT "" CACHE PATH "Owning FS150 SITL asset source")
if(NOT EXISTS "${FS150_ASSET_SOURCE_ROOT}/scripts/generate_native_flight_model.py")
  message(FATAL_ERROR "Set FS150_ASSET_SOURCE_ROOT; no copied physical defaults")
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/generated")
set(asset_header "${CMAKE_CURRENT_BINARY_DIR}/generated/fs150_native_flight_model.hpp")
set(asset_json "${CMAKE_CURRENT_BINARY_DIR}/generated/fs150_native_flight_model.json")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${FS150_ASSET_SOURCE_ROOT}/scripts/generate_native_flight_model.py"
  --sdf "${FS150_ASSET_SOURCE_ROOT}/models/fs150/iris.sdf"
  --params "${FS150_ASSET_SOURCE_ROOT}/config/generated/fs150-sitl.params"
  --fcu-name-catalog "${CMAKE_CURRENT_SOURCE_DIR}/fcu_parameter_names.json"
  --output "${asset_json}.tmp" --cpp-output "${asset_header}.tmp" RESULT_VARIABLE asset_status)
if(NOT asset_status EQUAL 0)
  message(FATAL_ERROR "FS150 asset extraction failed")
endif()
configure_file("${asset_header}.tmp" "${asset_header}" COPYONLY)
configure_file("${asset_json}.tmp" "${asset_json}" COPYONLY)
include_directories("${CMAKE_CURRENT_BINARY_DIR}/generated")
option(LIGHTWEIGHT_TESTS "Build the model and standalone server checks" OFF)

add_subdirectory(px4_v1_12_3)

find_package(Threads REQUIRED)
find_package(nlohmann_json 3.2 REQUIRED)
find_package(yaml-cpp REQUIRED)
find_package(XgcWorldLidar CONFIG REQUIRED)
find_package(catkin REQUIRED COMPONENTS roscpp geometry_msgs sensor_msgs nav_msgs std_srvs rosgraph_msgs mavros_msgs xgc2_lightweight_sim_msgs)
add_library(xsim_world STATIC xsim/world.cpp)
target_compile_features(xsim_world PUBLIC cxx_std_17)
target_include_directories(xsim_world PUBLIC "${XGC2_MATH_INCLUDE}" "${EIGEN_INCLUDE}" "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(xsim_world PUBLIC px4_v1_12_3::core XgcRoboticsInterfaces::Interfaces Threads::Threads nlohmann_json::nlohmann_json)
add_library(xsim_sensors STATIC xsim/sensors.cpp)
target_link_libraries(xsim_sensors PUBLIC xsim_world XgcWorldLidar::Geometry)
option(XSIM_GPU "Compile the explicit original GPU backend" OFF)
if(XSIM_GPU)
  target_compile_definitions(xsim_sensors PUBLIC XSIM_GPU=1)
  target_link_libraries(xsim_sensors PUBLIC XgcWorldLidar::Gpu)
endif()
add_executable(xsim xsim/main.cpp xsim/ros.cpp)
target_include_directories(xsim SYSTEM PRIVATE ${catkin_INCLUDE_DIRS})
target_link_libraries(xsim PRIVATE xsim_sensors yaml-cpp ${catkin_LIBRARIES})
foreach(target xsim_world xsim_sensors xsim)
  target_compile_options(${target} PRIVATE -O2 -Wall -Wextra)
endforeach()
if(LIGHTWEIGHT_TESTS)
  enable_testing()
  foreach(test_name rigid_body flight_controller model px4_setpoint ground_contact)
    add_executable(${test_name}_test ${test_name}_test.cpp)
    target_compile_options(${test_name}_test PRIVATE -O2 -Wall -Wextra -Werror -UNDEBUG)
    target_link_libraries(${test_name}_test PRIVATE xsim_world)
    add_test(NAME lightweight_${test_name} COMMAND ${test_name}_test)
  endforeach()
endif()
install(TARGETS xsim RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES "${asset_json}" DESTINATION ${CMAKE_INSTALL_DATADIR}/xsim)
install(DIRECTORY ../../config/ DESTINATION ${CMAKE_INSTALL_DATADIR}/xsim/config)
install(FILES ../../README.md DESTINATION ${CMAKE_INSTALL_DATADIR}/xsim)
install(DIRECTORY ../../docs/ DESTINATION ${CMAKE_INSTALL_DATADIR}/xsim/docs PATTERN "source-migration.json" EXCLUDE)
install(FILES px4_v1_12_3/upstream/LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/xsim/licenses RENAME PX4-LICENSE)
set(CPACK_PACKAGE_NAME "xsim")
set(CPACK_PACKAGE_VERSION "1.0.0")
set(CPACK_GENERATOR "TGZ")
include(CPack)

# Reference only in the private validation build. Never an installed runtime.
set(XSIM_BASELINE_SOURCE "" CACHE PATH "Pinned baseline source extracted by tests/validate.sh")
set(XSIM_BASELINE_SDK "" CACHE PATH "Read-only SDK headers for the historical reference fixture only")
if(LIGHTWEIGHT_TESTS AND XSIM_BASELINE_SOURCE)
  add_library(xsim_baseline_reference SHARED "${XSIM_BASELINE_SOURCE}/lightweight_vehicle.cpp")
  target_compile_options(xsim_baseline_reference PRIVATE -O2 -fvisibility=hidden)
  target_include_directories(xsim_baseline_reference PRIVATE "${XSIM_BASELINE_SDK}/abi/include" "${XSIM_BASELINE_SDK}/plugins/common" "${XSIM_BASELINE_SOURCE}/interfaces/include")
  target_include_directories(xsim_baseline_reference BEFORE PRIVATE
    "${XSIM_BASELINE_SOURCE}/px4_v1_12_3/include"
    "${XSIM_BASELINE_SOURCE}/px4_v1_12_3/upstream/src"
    "${XSIM_BASELINE_SOURCE}/px4_v1_12_3/upstream/src/lib"
    "${XSIM_BASELINE_SOURCE}/px4_v1_12_3/upstream/src/include"
    "${XSIM_BASELINE_SOURCE}/px4_v1_12_3/upstream/platforms/common/include")
  target_link_libraries(xsim_baseline_reference PRIVATE xsim_world)
  add_executable(xsim_replay ../../tests/replay.cpp)
  target_compile_options(xsim_replay PRIVATE -O2 -UNDEBUG)
  target_include_directories(xsim_replay PRIVATE "${XSIM_BASELINE_SDK}/abi/include" "${XSIM_BASELINE_SOURCE}/interfaces/include")
  target_link_libraries(xsim_replay PRIVATE xsim_world ${CMAKE_DL_LIBS})
  foreach(kind fs150 scout mecanum)
    add_test(NAME xsim_replay_${kind} COMMAND xsim_replay $<TARGET_FILE:xsim_baseline_reference> ${kind} ${CMAKE_CURRENT_BINARY_DIR}/${kind}-replay.json)
  endforeach()
endif()

if(LIGHTWEIGHT_TESTS)
  add_executable(xsim_world_test ../../tests/world_test.cpp)
  target_compile_options(xsim_world_test PRIVATE -O2 -UNDEBUG)
  target_link_libraries(xsim_world_test PRIVATE xsim_sensors)
  add_test(NAME xsim_world COMMAND xsim_world_test)
endif()
if(LIGHTWEIGHT_TESTS)
  add_executable(xsim_slow_service_test ../../tests/slow_service_test.cpp xsim/ros.cpp)
  target_compile_options(xsim_slow_service_test PRIVATE -O2 -UNDEBUG)
  target_include_directories(xsim_slow_service_test SYSTEM PRIVATE ${catkin_INCLUDE_DIRS})
  target_link_libraries(xsim_slow_service_test PRIVATE xsim_sensors ${catkin_LIBRARIES})
  set(XSIM_ADAPTER_SOURCE "" CACHE PATH "Read-only original Adapter source for the private projection fixture")
  if(XSIM_ADAPTER_SOURCE)
    add_executable(xsim_canonical_fixture ../../tests/canonical_fixture.cpp "${XSIM_ADAPTER_SOURCE}/common/src/localization_projection.cpp")
    target_compile_features(xsim_canonical_fixture PRIVATE cxx_std_17)
    target_include_directories(xsim_canonical_fixture PRIVATE "${XSIM_ADAPTER_SOURCE}/common/include" ${catkin_INCLUDE_DIRS})
    target_link_libraries(xsim_canonical_fixture PRIVATE ${catkin_LIBRARIES})
  endif()
endif()
option(XSIM_PRIVATE_ROS_TESTS "Run the private ROS integration/expansion fixture in CTest" OFF)
if(LIGHTWEIGHT_TESTS AND XSIM_PRIVATE_ROS_TESTS)
  if(NOT XSIM_BASELINE_SOURCE OR NOT XSIM_ADAPTER_SOURCE)
    message(FATAL_ERROR "Private ROS comparison requires pinned baseline and original Adapter projection sources")
  endif()
  add_test(NAME xsim_ros_contract COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/../../tests/ros_integration.py
    --xsim $<TARGET_FILE:xsim> --canonical-fixture $<TARGET_FILE:xsim_canonical_fixture>
    --slow-service-test $<TARGET_FILE:xsim_slow_service_test>
    --output ${CMAKE_CURRENT_BINARY_DIR}/ros-contract --replays ${CMAKE_CURRENT_BINARY_DIR})
  set_tests_properties(xsim_ros_contract PROPERTIES DEPENDS "xsim_replay_fs150;xsim_replay_scout;xsim_replay_mecanum;xsim_world" TIMEOUT 180)
endif()
